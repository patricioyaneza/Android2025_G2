-- Ejercicio 5: id, nombre, apellido, nacimiento, ciudad, genero, id prov, nom prov y alergias
-- Ejercicio 6: si la alergia es null que muestre Sin Alergia (investigar)
select pa.patient_id as 'ID Paciente', 
concat(pa.first_name,' ', pa.last_name) as 'Nombre Paciente', 
pa.birth_date as 'Fecha de Nacimiento', 
pa.city as Ciudad, 
pa.gender as Genero, 
pa.province_id as 'Código Provincia', 
pr.province_name as 'Nombre Provincia', 
coalesce(allergies, 'Sin Alergia') as Alergias -- ifnull
from patients pa
	INNER JOIN province_names pr on pr.province_id = pa.province_id



-- Ejercicio 7: Mostrar los datos de atención y el paciente atendido
SELECT  strftime('%d-%m-%Y', ad.admission_date) AS 'Fecha Ingreso',
        ad.diagnosis AS Diagnostico,
        pa.patient_id AS ID,
        pa.first_name AS Nombre,
        pa.last_name AS Apellido,
        strftime('%d-%m-%Y', pa.birth_date) AS Nacimiento,
        pa.city AS Ciudad,
        pa.gender AS Genero,
        pr.province_name AS 'Provincia',
        COALESCE(pa.allergies, 'sin alergias') AS Alergias

  --      ad.attending_doctor_id AS 'ID Doctor'
FROM patients pa
	INNER JOIN province_names pr ON pr.province_id = pa.province_id
	INNER JOIN admissions ad ON ad.patient_id = pa.patient_id;


-- Ejercicio 8: Ordenar por fecha de llegada y modificar el diagnostico mostrandolo en mayuscula
-- Ejercicio 9: Mostrar las atenciones y los datos de los pacientes en orden de llegada y diagnostico en mayúscula 
y el doctor que atendió. Mostrar las columnas en español
SELECT  strftime('%d-%m-%Y', ad.admission_date) AS 'Fecha Ingreso',
        upper(ad.diagnosis) AS Diagnostico,
        CONCAT(do.first_name,' ',do.last_name) AS 'NOMBRE COMPLETO DOCTOR',
        pa.patient_id AS ID,
        pa.first_name AS Nombre,
        pa.last_name AS Apellido,
        strftime('%d-%m-%Y', pa.birth_date) AS Nacimiento,
        pa.city AS Ciudad,
        pa.gender AS Genero,
        pr.province_name AS 'Provincia',
        COALESCE(pa.allergies, 'sin alergias') AS Alergias

  --      ad.attending_doctor_id AS 'ID Doctor'
FROM patients pa
	INNER JOIN province_names pr ON pr.province_id = pa.province_id
	INNER JOIN admissions ad ON ad.patient_id = pa.patient_id
    INNER JOIN doctors do ON ad.attending_doctor_id =do.doctor_id
order by ad.admission_date

-- EJERCICIO 11: contar y agrupar pacientes de acuerdo a la PROVINCIA de origen para la ciudad de Ontario

SELECT pn.province_name as Provincia,
	pa.city as Ciudad,
	COUNT(pa.patient_id) AS 'N° Pacientes'
FROM patients pa
	inner JOIN province_names pn ON pa.province_id = pn.province_id
where pn.province_name ='Ontario'
GROUP BY pn.province_name, pa.city
ORDER BY COUNT(pa.patient_id) DESC, pn.province_name,Ciudad;


-- Ejercicio 12: Contar la cantidad de pacientes con alérgias a algun compuesto, ordenar alfabeticamente.

SELECT allergies AS Alergia,
		COUNT(patient_id) AS 'N° Pacientes',
        concat(((COUNT(patient_id) * 100) / ( select count(*) from patients )), '%') Porcentaje
FROM patients
WHERE allergies IS NOT NULL AND trim(allergies) != ''
GROUP BY allergies
ORDER BY 2 desc;

-- Ejercicio 13: top 10 de alergias que tienen los pacientes de la ciudad de Hamilton

SELECT pa.allergies AS Alergia,
		pa.city as Ciudad,
      	COUNT(pa.patient_id) AS "Nro pacientes",
      	ROUND( COUNT(pa.patient_id) * 100.0 / (SELECT COUNT(*) FROM patients ), 2) AS porcentaje
FROM patients pa
WHERE pa.allergies IS NOT NULL
and pa.city = 'Hamilton'
GROUP BY alergia
ORDER BY porcentaje DESC
limit 10;
